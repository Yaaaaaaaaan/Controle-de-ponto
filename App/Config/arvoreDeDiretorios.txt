 -------------------------- ------------------ -----------------           # Diagrama da lógica implementada em IndexDB
PCDB/                                         # Nome do Banco de Dados no IndexedDB
├── Object Store: userData
│   ├── Chave Primária: 'nickname'
│   └── Índices:
│       └── userId (no campo 'userId')
│
├── Object Store: userToken
│   ├── Chave Primária: 'token'
│   └── Índices:
│       ├── userId (no campo 'userId')
│       └── lastUpdated (no campo 'lastUpdated')
│
├── Object Store: userPictures
│   ├── Chave Primária: 'picId'
│   └── Índices:
│       └── userId (no campo 'userId')
│
├── Object Store: pointControl
│   ├── Chave Primária: 'cod' (auto-incremento)
│   └── Índices:
│       ├── userId (no campo 'userId')
│       ├── status (no campo 'status')
│       └── dateInIdx (no campo 'dateIn')
│
└── Object Store: syncQueue
    ├── Chave Primária: 'id' (auto-incremento)
    └── Índices:
        └── timestamp (no campo 'timestamp')
